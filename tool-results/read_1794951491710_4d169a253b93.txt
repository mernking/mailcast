   111→/**
   112→ * Best-effort conversion of an HTML body string to a markdown approximation.
   113→ * Not exhaustive — covers headings, lists, strong/em, links, line breaks and
   114→ * paragraphs. Anything unrecognized is stripped of its tags. Used when the
   115→ * user switches from a Visual/HTML view into the Markdown editor so the
   116→ * content carries over in a readable shape.
   117→ */
   118→function htmlToMarkdown(html: string): string {
   119→  if (!html) return "";
   120→  // Normalize whitespace once, then walk through tag-based replacements.
   121→  let s = html
   122→    // Block-level boundaries first so subsequent replacements stay ordered.
   123→    .replace(/<\/(h[1-6]|p|div|li|tr|blockquote)>/gi, "\n")
   124→    .replace(/<br\s*\/?>/gi, "\n")
   125→    // Headings → # / ## / ### etc.
   126→    .replace(/<h1[^>]*>([\s\S]*?)<\/h1>/gi, (_m, t) => `# ${stripTags(t).trim()}\n`)
   127→    .replace(/<h2[^>]*>([\s\S]*?)<\/h2>/gi, (_m, t) => `## ${stripTags(t).trim()}\n`)
   128→    .replace(/<h3[^>]*>([\s\S]*?)<\/h3>/gi, (_m, t) => `### ${stripTags(t).trim()}\n`)
   129→    .replace(/<h[4-6][^>]*>([\s\S]*?)<\/h[4-6]>/gi, (_m, t) => `#### ${stripTags(t).trim()}\n`)
   130→    // Inline emphasis.
   131→    .replace(/<(strong|b)\b[^>]*>([\s\S]*?)<\/\1>/gi, (_m, _t, c) => `**${c.trim()}**`)
   132→    .replace(/<(em|i)\b[^>]*>([\s\S]*?)<\/\1>/gi, (_m, _t, c) => `*${c.trim()}*`)
   133→    .replace(/<code\b[^>]*>([\s\S]*?)<\/code>/gi, (_m, c) => `\`${c.trim()}\``)
   134→    // Links: [text](href)
   135→    .replace(/<a\b[^>]*href=["']([^"']*)["'][^>]*>([\s\S]*?)<\/a>/gi, (_m, href, text) => `[${stripTags(text).trim()}](${href})`)
   136→    // Images: ![alt](src)
   137→    .replace(/<img\b[^>]*alt=["']([^"']*)["'][^>]*src=["']([^"']*)["'][^>]*\/?>/gi, (_m, alt, src) => `![${alt}](${src})`)
   138→    .replace(/<img\b[^>]*src=["']([^"']*)["'][^>]*\/?>/gi, (_m, src) => `![](${src})`)
   139→    // Lists: bullets first, then numbered. <ul>/<ol> tags become newlines.
   140→    .replace(/<ul[^>]*>/gi, "\n")
   141→    .replace(/<\/ul>/gi, "\n")
   142→    .replace(/<ol[^>]*>/gi, "\n")
   143→    .replace(/<\/ol>/gi, "\n")
   144→    .replace(/<li\b[^>]*>([\s\S]*?)<\/li>/gi, (_m, c) => `- ${c.trim()}\n`)
   145→    // Blockquote lines.
   146→    .replace(/<blockquote[^>]*>([\s\S]*?)<\/blockquote>/gi, (_m, c) =>
   147→      stripTags(c).trim().split("\n").map((l: string) => `> ${l}`).join("\n") + "\n")
   148→    // Horizontal rules.
   149→    .replace(/<hr\s*\/?>/gi, "\n---\n")
   150→    // Drop any remaining tags we haven't converted.
   151→    .replace(/<[^>]+>/g, "")
   152→    // Decode a few common entities.
   153→    .replace(/&nbsp;/g, " ")
   154→    .replace(/&amp;/g, "&")
   155→    .replace(/&lt;/g, "<")
   156→    .replace(/&gt;/g, ">")
   157→    .replace(/&quot;/g, '"')
   158→    .replace(/&#39;/g, "'")
   159→    // Tidy whitespace.
   160→    .replace(/\r\n/g, "\n")
   161→    .replace(/[ \t]+\n/g, "\n")
   162→    .replace(/\n{3,}/g, "\n\n")
   163→    .trim();
   164→  return s;
   165→}
   166→
   167→function stripTags(html: string): string {
   168→  return String(html).replace(/<[^>]+>/g, "");
   169→}

   171→export function CampaignEditor({ campaignId }: { campaignId: string }) {
   172→  const isNew = campaignId === "new";
   173→  const openCampaignEditor = useAppStore((s) => s.openCampaignEditor);
   174→  const setSection = useAppStore((s) => s.setSection);
   175→
   176→  const campaignUrl = isNew ? null : `/api/campaigns/${campaignId}`;
   177→  const { data: campaign, loading, error } = useFetch<CampaignDetail>(campaignUrl);
   178→  const { data: senders } = useFetch<Sender[]>("/api/senders");
   179→  const { data: statsOverview } = useFetch<{ activeContacts: number }>("/api/stats");
   180→
   181→  const [title, setTitle] = useState("");
   182→  const [subject, setSubject] = useState("");
   183→  const [preheader, setPreheader] = useState("");
   184→  const [content, setContent] = useState("");
   185→  const [contentType, setContentType] = useState<string>("html");
   186→  const [senderId, setSenderId] = useState("");
   187→  const [hydrated, setHydrated] = useState(false);
   188→
   189→  const [saving, setSaving] = useState(false);
   190→  const [sending, setSending] = useState(false);
   191→  const [sendDialogOpen, setSendDialogOpen] = useState(false);
   192→  const [mode, setMode] = useState<EditorMode>("visual");
   193→  const htmlTextareaRef = useRef<HTMLTextAreaElement>(null);
   194→  const mdTextareaRef = useRef<HTMLTextAreaElement>(null);
   195→
   196→  // Populate fields when the campaign loads (or when switching to "new").
   197→  useEffect(() => {
   198→    if (campaign) {
   199→      setTitle(campaign.title || "");
   200→      setSubject(campaign.subject || "");
   201→      setPreheader(campaign.preheader || "");
   202→      setContent(campaign.content || "");
   203→      const ct = campaign.contentType === "html" ? "html" : "markdown";
   204→      setContentType(ct);
   205→      // Default to the matching editor view for the loaded content type.
   206→      setMode(ct === "html" ? "visual" : "markdown");
   207→    } else if (isNew) {
   208→      setTitle("");
   209→      setSubject("");
   210→      setPreheader("");
   211→      setContent("");
   212→      setContentType("html");
   213→      setMode("visual");
   214→      setSenderId("");
   215→    }
   216→    setHydrated(true);
   217→  }, [campaign, isNew]);
   218→
   219→  // Stats panel — only relevant when the campaign has been (or is being) sent.
   220→  const liveStatus = campaign?.status;
   221→  const statsUrl =
   222→    !isNew && (liveStatus === "sent" || liveStatus === "sending")
   223→      ? `/api/campaigns/${campaignId}/stats`
   224→      : null;
   225→  const {
   226→    data: stats,
   227→    refetch: refetchStats,
   228→  } = useFetch<CampaignStats>(statsUrl);
   229→
   230→  // While the campaign is actively sending, poll stats every 3 seconds.
   231→  useEffect(() => {
   232→    if (liveStatus !== "sending") return;
   233→    const t = setInterval(() => {
   234→      refetchStats();
   235→    }, 3000);
   236→    return () => clearInterval(t);
   237→  }, [liveStatus, refetchStats]);
   238→
   239→  const canEdit =
   240→    !campaign ||
   241→    campaign.status === "draft" ||
   242→    campaign.status === "failed";
   243→
   244→  /**
   245→   * Insert a merge tag at the cursor of whichever textarea is currently the
   246→   * active editor (HTML or Markdown). For the Visual editor we rely on the
   247→   * contentEditable's own selection and execCommand("insertText").
   248→   */
   249→  const insertMergeTagTextarea = (
   250→    ref: React.RefObject<HTMLTextAreaElement | null>,
   251→    tag: string
   252→  ) => {
   253→    const el = ref.current;
   254→    if (!el) {
   255→      setContent((c) => c + tag);
   256→      return;
   257→    }
   258→    const start = el.selectionStart ?? content.length;
   259→    const end = el.selectionEnd ?? content.length;
   260→    const next = content.slice(0, start) + tag + content.slice(end);
   261→    setContent(next);
   262→    requestAnimationFrame(() => {
   263→      el.focus();
   264→      const pos = start + tag.length;
   265→      el.setSelectionRange(pos, pos);
   266→    });
   267→  };
   268→
   269→  const insertMergeTag = (tag: string) => {
   270→    if (mode === "html") {
   271→      insertMergeTagTextarea(htmlTextareaRef, tag);
   272→    } else if (mode === "markdown") {
   273→      insertMergeTagTextarea(mdTextareaRef, tag);
   274→    } else if (mode === "visual") {
   275→      // For the contentEditable editor: try execCommand("insertText"),
   276→      // which preserves the caret position. Fallback: append to content.
   277→      try {
   278→        const ok = document.execCommand("insertText", false, tag);
   279→        if (ok) return;
   280→      } catch {
   281→        /* fall through */
   282→      }
   283→      setContent((c) => c + tag);
   284→    }
   285→  };
   286→
   287→  /**
   288→   * Handle mode switches. When moving between HTML and Markdown formats we
   289→   * convert the underlying `content` so nothing is lost. Visual ↔ HTML is
   290→   * a pure view change (both edit the same HTML string).
   291→   */
   292→  const handleModeChange = (next: string) => {
   293→    const nextMode = next as EditorMode;
   294→    if (nextMode === mode) return;
   295→
   296→    if (nextMode === "visual" || nextMode === "html") {
   297→      // Target is HTML.
   298→      if (contentType === "markdown") {
   299→        const html = markdownToHtml(content);
   300→        setContent(html);
   301→        setContentType("html");
   302→      }
   303→      // If already html, nothing to convert — just switch view.
   304→    } else if (nextMode === "markdown") {
   305→      if (contentType === "html") {
   306→        const md = htmlToMarkdown(content);
   307→        setContent(md);
   308→        setContentType("markdown");
   309→      }
   310→      // If already markdown, nothing to convert.
   311→    }
   312→    // "preview" never converts — it reads content as-is.
   313→    setMode(nextMode);
   314→  };
   315→
   316→  const handleSave = async () => {
   317→    if (!title.trim()) return toast.error("Add a title first.");
   318→    if (!subject.trim()) return toast.error("Add a subject first.");
   319→    if (!content.trim()) return toast.error("Add some content first.");
   320→    if (!senderId) return toast.error("Pick a sender first.");
   321→
   322→    setSaving(true);
   323→    try {
   324→      const payload = {
   325→        title: title.trim(),
   326→        subject: subject.trim(),
   327→        content,
   328→        contentType,
   329→        preheader: preheader.trim() || null,
   330→        senderId,
   331→      };
   332→      if (isNew) {
   333→        const created = await apiFetch<CampaignDetail>("/api/campaigns", {
   334→          method: "POST",
   335→          body: JSON.stringify(payload),
   336→        });
   337→        toast.success("Draft saved");
   338→        openCampaignEditor(created.id);
   339→      } else {
   340→        await apiFetch<CampaignDetail>(`/api/campaigns/${campaignId}`, {
   341→          method: "PATCH",
   342→          body: JSON.stringify(payload),
   343→        });
   344→        toast.success("Draft saved");
   345→      }
   346→    } catch (err) {
   347→      toast.error(err instanceof Error ? err.message : "Failed to save draft");
   348→    } finally {
   349→      setSaving(false);
   350→    }
   351→  };
   352→
   353→  const handleSend = async () => {
   354→    setSendDialogOpen(false);
   355→    if (isNew) {
   356→      toast.error("Save the draft first before sending.");
   357→      return;
   358→    }
   359→    setSending(true);
   360→    try {
   361→      const result = await apiFetch<{
   362→        ok: boolean;
   363→        totalRecip: number;
   364→        sentCount: number;
   365→        failedCount: number;
   366→        warning?: string;
   367→      }>(`/api/campaigns/${campaignId}/send`, { method: "POST" });
   368→      toast.success(
   369→        `Broadcast sent: ${result.sentCount} delivered, ${result.failedCount} failed.`,
   370→        { description: result.warning }
   371→      );
   372→      // Force a full refresh of campaign + stats.
   373→      window.location.reload();
   374→    } catch (err) {
   375→      toast.error(err instanceof Error ? err.message : "Broadcast failed");
   376→    } finally {
   377→      setSending(false);
   378→    }
   379→  };
   380→
   381→  if (loading) {
   382→    return (
   383→      <div className="max-w-4xl mx-auto w-full flex flex-col gap-6">
   384→        <Skeleton className="h-9 w-40" />
   385→        <Skeleton className="h-72 rounded-2xl" />
   386→        <Skeleton className="h-64 rounded-2xl" />
   387→      </div>
   388→    );
   389→  }
   390→
   391→  if (error) {
   392→    return (
   393→      <div className="max-w-2xl mx-auto w-full">
   394→        <Alert variant="destructive">
   395→          <AlertTriangle />
   396→          <AlertTitle>Could not load campaign</AlertTitle>
   397→          <AlertDescription>{error}</AlertDescription>
   398→        </Alert>
   399→        <Button
   400→          variant="ghost"
   401→          className="mt-4"
   402→          onClick={() => openCampaignEditor(null)}
   403→        >
   404→          <ArrowLeft className="size-4" /> Back to campaigns
   405→        </Button>
   406→      </div>
   407→    );
   408→  }
   409→
   410→  const senderList = senders || [];
   411→  const activeContacts = statsOverview?.activeContacts ?? 0;
   412→  const isSending = liveStatus === "sending";
   413→  const isSent = liveStatus === "sent";
   414→  const showStatsPanel = !isNew && (isSending || isSent) && !!stats;
   415→
   416→  const sendDisabled =
   417→    isNew ||
   418→    !senderId ||
   419→    !content.trim() ||
   420→    !subject.trim() ||
   421→    activeContacts === 0 ||
   422→    isSending ||
   423→    isSent;
   424→
   425→  return (
   426→    <div className="max-w-4xl mx-auto w-full flex flex-col gap-6">
   427→      <div className="flex items-center justify-between gap-3">
   428→        <Button
   429→          variant="ghost"
   430→          size="sm"
   431→          className="rounded-full"
   432→          onClick={() => openCampaignEditor(null)}
   433→        >
   434→          <ArrowLeft className="size-4" /> All campaigns
   435→        </Button>
   436→        {campaign && (
   437→          <div className="flex items-center gap-2">
   438→            <StatusBadge status={campaign.status} />
   439→          </div>
   440→        )}
   441→      </div>
   442→
   443→      {/* Stats panel — visible once sending/sent */}
   444→      {showStatsPanel && stats ? (
   445→        <StatsPanel stats={stats} isSending={isSending} />
   446→      ) : null}
   447→
   448→      {/* Sender warning */}
   449→      {senderList.length === 0 && (
   450→        <Alert>
   451→          <MailWarning />
   452→          <AlertTitle>No sender connected</AlertTitle>
   453→          <AlertDescription>
   454→            You need to add a Gmail sender before you can broadcast.{" "}
   455→            <button
   456→              className="underline font-medium"
   457→              onClick={() => setSection("senders")}
   458→            >
   459→              Add a sender
   460→            </button>
   461→            .
   462→          </AlertDescription>
   463→        </Alert>
   464→      )}
   465→
   466→      {/* No recipients warning */}
   467→      {activeContacts === 0 && hydrated && (
   468→        <Alert>
   469→          <MailWarning />
   470→          <AlertTitle>No active contacts</AlertTitle>
   471→          <AlertDescription>
   472→            You have no active contacts to send to.{" "}
   473→            <button
   474→              className="underline font-medium"
   475→              onClick={() => setSection("contacts")}
   476→            >
   477→              Add contacts
   478→            </button>
   479→            .
   480→          </AlertDescription>
   481→        </Alert>
   482→      )}
   483→
   484→      {/* Header fields */}
   485→      <div className="bg-card border rounded-2xl p-5 flex flex-col gap-4">
   486→        <div className="flex flex-col gap-2">
   487→          <Label htmlFor="campaign-title" className="text-xs uppercase tracking-wide text-muted-foreground">
   488→            Title
   489→          </Label>
   490→          <Input
   491→            id="campaign-title"
   492→            value={title}
   493→            onChange={(e) => setTitle(e.target.value)}
   494→            placeholder="Q1 product update"
   495→            disabled={!canEdit}
   496→          />
   497→        </div>
   498→        <div className="grid sm:grid-cols-2 gap-4">
   499→          <div className="flex flex-col gap-2">
   500→            <Label htmlFor="campaign-subject" className="text-xs uppercase tracking-wide text-muted-foreground">
   501→              Subject
   502→            </Label>
   503→            <Input
   504→              id="campaign-subject"
   505→              value={subject}
   506→              onChange={(e) => setSubject(e.target.value)}
   507→              placeholder="What we shipped this month"
   508→              disabled={!canEdit}
   509→            />
   510→          </div>
   511→          <div className="flex flex-col gap-2">
   512→            <Label htmlFor="campaign-preheader" className="text-xs uppercase tracking-wide text-muted-foreground">
   513→              Preheader
   514→            </Label>
   515→            <Input
   516→              id="campaign-preheader"
   517→              value={preheader}
   518→              onChange={(e) => setPreheader(e.target.value)}
   519→              placeholder="A quick peek inside this email"
   520→              disabled={!canEdit}
   521→            />
   522→            <p className="text-[11px] text-muted-foreground">
   523→              Shown in inbox preview next to the subject.
   524→            </p>
   525→          </div>
   526→        </div>
   527→        <div className="flex flex-col gap-2">
   528→          <Label htmlFor="campaign-sender" className="text-xs uppercase tracking-wide text-muted-foreground">
   529→            Sender
   530→          </Label>
   531→          <Select
   532→            value={senderId}
   533→            onValueChange={setSenderId}
   534→            disabled={!canEdit || senderList.length === 0}
   535→          >
   536→            <SelectTrigger id="campaign-sender" className="w-full">
   537→              <SelectValue placeholder={senderList.length === 0 ? "No senders added" : "Pick a sender"} />
   538→            </SelectTrigger>
   539→            <SelectContent>
   540→              {senderList.map((s) => (
   541→                <SelectItem key={s.id} value={s.id}>
   542→                  {s.fromName} &lt;{s.email}&gt;{!s.verified ? " · Not verified" : ""}
   543→                </SelectItem>
   544→              ))}
   545→            </SelectContent>
   546→          </Select>
   547→        </div>
   548→      </div>
   549→
   550→      {/* Body: Visual / HTML / Markdown / Preview */}
   551→      <div className="bg-card border rounded-2xl p-5 flex flex-col gap-4">
   552→        <Tabs value={mode} onValueChange={handleModeChange}>
   553→          <div className="flex items-center justify-between gap-3 flex-wrap">
   554→            <div>
   555→              <h3 className="font-serif text-lg text-foreground">Body</h3>
   556→              <p className="text-xs text-muted-foreground">
   557→                Write visually, hand-edit HTML, or use Markdown. Use merge tags to personalize per recipient.
   558→              </p>
   559→            </div>
   560→            <TabsList className="flex-wrap h-auto">
   561→              <TabsTrigger value="visual" disabled={!canEdit} className="gap-1.5">
   562→                <PencilLine className="size-3.5" /> Visual
   563→              </TabsTrigger>
   564→              <TabsTrigger value="html" disabled={!canEdit} className="gap-1.5">
   565→                <Code className="size-3.5" /> HTML
   566→              </TabsTrigger>
   567→              <TabsTrigger value="markdown" disabled={!canEdit} className="gap-1.5">

   761→  subject,
   762→  preheader,
   763→  content,
   764→  contentType,
   765→  fromName,

   768→  preheader: string;

   804→function StatsPanel({
   805→  stats,
   806→  isSending,
   807→}: {
   808→  stats: CampaignStats;
   809→  isSending: boolean;
   810→}) {
   811→  const total = stats.totalRecip || 1;
   812→  const percent = Math.round(((stats.sentCount + stats.failedCount) / total) * 100);
   813→  return (
   814→    <div className="bg-card border rounded-2xl p-5 flex flex-col gap-4">
   815→      <div className="flex items-center justify-between">
   816→        <div>
   817→          <h3 className="font-serif text-lg text-foreground">Delivery</h3>
   818→          <p className="text-xs text-muted-foreground">
   819→            {isSending ? "Currently sending…" : "Broadcast complete"}
   820→          </p>
   821→        </div>
   822→        <div className="text-right">
   823→          <div className="font-serif text-2xl text-foreground">{percent}%</div>
   824→          <div className="text-xs text-muted-foreground">
   825→            {stats.sentCount + stats.failedCount} / {stats.totalRecip}
   826→          </div>
   827→        </div>
   828→      </div>
   829→
   830→      <Progress value={percent} />
   831→
   832→      <div className="grid grid-cols-3 gap-3">
   833→        <StatTile label="Sent" value={stats.sentCount} tone="success" />
   834→        <StatTile label="Failed" value={stats.failedCount} tone="danger" />
   835→        <StatTile label="Pending" value={stats.pendingCount} tone="neutral" />
   836→      </div>
   837→
   838→      {stats.recent.length > 0 && (
   839→        <div className="mt-2">
   840→          <h4 className="text-sm font-medium text-foreground mb-2">
   841→            Recent deliveries
   842→          </h4>
   843→          <div className="max-h-72 overflow-y-auto scroll-thin border rounded-lg">
   844→            <Table>
   845→              <TableHeader>
   846→                <TableRow>
   847→                  <TableHead className="pl-3">Email</TableHead>
   848→                  <TableHead>Status</TableHead>
   849→                  <TableHead>Error</TableHead>
   850→                  <TableHead className="pr-3 text-right">Sent at</TableHead>
   851→                </TableRow>
   852→              </TableHeader>
   853→              <TableBody>
   854→                {stats.recent.map((r, i) => (
   855→                  <TableRow key={`${r.email}-${i}`}>
   856→                    <TableCell className="pl-3 font-medium max-w-[200px] truncate">
   857→                      {truncate(r.email, 36)}
   858→                    </TableCell>
   859→                    <TableCell>
   860→                      <RecipientStatusBadge status={r.status} />
   861→                    </TableCell>
   862→                    <TableCell className="text-xs text-muted-foreground max-w-[200px] truncate">
   863→                      {r.error || "—"}
   864→                    </TableCell>

   866→                      {fmtDateTime(r.sentAt)}
   867→                    </TableCell>
   868→                  </TableRow>
   869→                ))}
   870→              </TableBody>
   871→            </Table>
   872→          </div>
   873→        </div>
   874→      )}
   875→    </div>
   876→  );
